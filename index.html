#include <WiFi.h>
#include <WiFiUdp.h>
#include <driver/i2s.h>

const char* WIFI_SSID = "YOUR_WIFI_NETWORK";
const char* WIFI_PASS = "YOUR_WIFI_PASSWORD";
const char* LAPTOP_IP = "192.168.1.100"; // REPLACE WITH PERSON 2'S IP
const int UDP_PORT = 12345;

#define I2S_MIC_SCK 2
#define I2S_MIC_WS  42
#define I2S_MIC_SD  41

WiFiUDP udp;
#define DMA_BUF_LEN 512
int16_t audio_buffer[DMA_BUF_LEN];

void setup() {
  Serial.begin(115200);
  i2s_config_t i2s_config = {
    .mode = (i2s_mode_t)(I2S_MODE_MASTER | I2S_MODE_RX),
    .sample_rate = 16000,
    .bits_per_sample = I2S_BITS_PER_SAMPLE_16BIT,
    .channel_format = I2S_CHANNEL_FMT_ONLY_LEFT,
    .communication_format = I2S_COMM_FORMAT_STAND_I2S,
    .intr_alloc_flags = ESP_INTR_FLAG_LEVEL1,
    .dma_buf_count = 4,
    .dma_buf_len = DMA_BUF_LEN,
  };
  i2s_pin_config_t pin_config = {
    .bck_io_num = I2S_MIC_SCK,
    .ws_io_num = I2S_MIC_WS,
    .data_out_num = I2S_PIN_NO_CHANGE,
    .data_in_num = I2S_MIC_SD
  };
  i2s_driver_install(I2S_NUM_0, &i2s_config, 0, NULL);
  i2s_set_pin(I2S_NUM_0, &pin_config);

  WiFi.begin(WIFI_SSID, WIFI_PASS);
  while (WiFi.status() != WL_CONNECTED) { delay(500); }
}

void loop() {
  size_t bytes_read = 0;
  esp_err_t result = i2s_read(I2S_NUM_0, audio_buffer, sizeof(audio_buffer), &bytes_read, portMAX_DELAY);
  
  if (result == ESP_OK && bytes_read > 0) {
    udp.beginPacket(LAPTOP_IP, UDP_PORT);
    udp.write((uint8_t*)audio_buffer, bytes_read);
    udp.endPacket();
  }
}

  ring.begin();
  ring.setBrightness(80);
  ring.fill(ring.Color(0, 50, 100)); // Dim Cyan
  ring.show();

  // Config I2S for TX omitted for brevity (match doc specs)
  
  WiFi.begin(WIFI_SSID, WIFI_PASS);
  while (WiFi.status() != WL_CONNECTED) { delay(100); }

  audioServer.begin();
  wsClient.begin(LAPTOP_IP, 8081, "/");
}

void loop() {
  wsClient.loop();
  WiFiClient client = audioServer.available();
  if (client) {
    uint8_t audioData[512];
    size_t bytesWritten;
    while (client.connected()) {
      while (client.available()) {
        int bytesRead = client.read(audioData, sizeof(audioData));
        if (bytesRead > 0) {
          i2s_write(I2S_NUM_1, audioData, bytesRead, &bytesWritten, portMAX_DELAY);
        }
      }
    }
    client.stop();
  }
}

def capture_enrollment(user_name, filename):
    print(f"\n--- CALIBRATING {user_name.upper()} ---")
    recordings = []
    
    for i in range(3):
        input(f"Press Enter to record phrase {i+1}/3 ('I am an authorized user of this system')...")
        audio = sd.rec(int(4 * 16000), samplerate=16000, channels=1, dtype='float32')
        sd.wait()
        recordings.append(torch.tensor(audio.T))
        print("Captured.")
    
    embeddings = [spk_verifier.encode_batch(rec) for rec in recordings]
    master_embedding = torch.mean(torch.stack(embeddings), dim=0)
    torch.save(master_embedding, filename)
    print(f"Vector saved to {filename}")

if __name__ == "__main__":

Phase 4: Fabrication & Integration

SAFELY HOUSE ELECTRONICS & TEST LOOP

4.1 Mechanical Assembly

  • Outer Bezel / Arc Reactor Construct three-layer circular structure. House 24-LED ring behind a 100mm optical diffusion disc.
  • Acoustic Isolation Mount INMP441 strictly to the upper rim in a silicone grommet facing upward, totally separate from speakers.
  • Resonance Chambers The two series-wired speakers must sit in sealed acoustic pods to prevent airwave cancellation.
  • Thermal Backplate Mount both ESP32-S3 boards with a physical air gap between chips and the rear plastic shield against the chest.

4.2 Safe Power Management

Do not use raw Li-Po cells. Route a cut USB cable from the 5V power bank. Run positive line through the SPDT slide switch. Check the 1000µF capacitor is tightly seated before flipping switch.

4.3 Final System Checkout

Execute checklist prior to live demonstration:

  • Laptop Python bridge (server.py) running on local network.
  • SPDT slide switch powered on; LEDs pulse Dim Cyan (Standby).
  • Person 1 says, "Hey JARVIS." LEDs turn bright Cyan immediately.
  • Person 1 asks, "What is the equivalent resistance formula for a parallel circuit?"
  • Biometric Check: LEDs spin Amber. Similarity >= 0.72 turns LEDs Green.
  • Gemini processes audio (LEDs Violet). JARVIS replies through speakers.
  • Third party says "Hey JARVIS, reset system." LEDs flash Red three times (Lockout), ignored.
